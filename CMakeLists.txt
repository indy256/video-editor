cmake_minimum_required(VERSION 3.24)
project(VideoEditor VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Network Test)
add_library(editor_core src/timeline.cpp src/jobs.cpp src/editor.cpp src/rangebridge.cpp src/chunkcache.cpp)
target_include_directories(editor_core PUBLIC src)
target_link_libraries(editor_core PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Network)
qt_add_executable(VideoEditor WIN32 src/main.cpp)
target_link_libraries(VideoEditor PRIVATE editor_core)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(editor_tests tests/editor_tests.cpp)
  set_target_properties(editor_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(editor_tests PRIVATE editor_core Qt6::Test)
  add_test(NAME editor_tests COMMAND editor_tests -platform offscreen -o ${CMAKE_CURRENT_BINARY_DIR}/test-results.txt,txt)
  set_tests_properties(editor_tests PROPERTIES TIMEOUT 300)
endif()
