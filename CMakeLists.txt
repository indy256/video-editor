cmake_minimum_required(VERSION 3.24)
project(VideoEditor VERSION 1.0.0 LANGUAGES CXX)
set(VIDEO_EDITOR_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version")
string(REGEX REPLACE "^v" "" VIDEO_EDITOR_VERSION "${VIDEO_EDITOR_VERSION}")
if(NOT VIDEO_EDITOR_VERSION)
  set(VIDEO_EDITOR_VERSION "${PROJECT_VERSION}")
endif()
if(NOT VIDEO_EDITOR_VERSION MATCHES "^[0-9]+[.][0-9]+[.][0-9]+([-+][0-9A-Za-z.+-]+)?$")
  message(FATAL_ERROR "Invalid release version: ${VIDEO_EDITOR_VERSION}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(cmake/ReleaseOptimizations.cmake)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Network)
option(VIDEO_EDITOR_REQUIRE_QT_LTO "Require the verified Qt LTO kit" OFF)
if(VIDEO_EDITOR_REQUIRE_QT_LTO)
  get_filename_component(qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
  file(READ "${qt_prefix}/lto-build.json" qt_lto_manifest)
  foreach(module qtbase qtshadertools qtsvg qtdeclarative qtmultimedia)
    string(JSON module_lto GET "${qt_lto_manifest}" modules ${module} lto)
    if(NOT module_lto)
      message(FATAL_ERROR "Qt module ${module} was not built with LTO")
    endif()
  endforeach()
endif()
add_library(editor_core src/timeline.cpp src/jobs.cpp src/editor.cpp src/rangebridge.cpp src/chunkcache.cpp)
target_include_directories(editor_core PUBLIC src)
target_link_libraries(editor_core PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Network)
if(MSVC)
  target_compile_options(editor_core PUBLIC /utf-8)
endif()
qt_add_executable(VideoEditor WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(VideoEditor PRIVATE editor_core)
set_target_properties(VideoEditor PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER com.github.indy256.video-editor
  MACOSX_BUNDLE_BUNDLE_NAME "Video Editor"
  MACOSX_BUNDLE_BUNDLE_VERSION "${VIDEO_EDITOR_VERSION}"
  MACOSX_BUNDLE_SHORT_VERSION_STRING "${VIDEO_EDITOR_VERSION}")

option(VIDEO_EDITOR_DEPLOY "Install the application, Qt runtime and helpers" OFF)
if(VIDEO_EDITOR_DEPLOY)
  include(GNUInstallDirs)
  set(CMAKE_INSTALL_LIBDIR lib)
  set(VIDEO_EDITOR_TOOLS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tools" CACHE PATH "Directory containing release helpers")
  install(TARGETS VideoEditor BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES README.md packaging/THIRD_PARTY.md DESTINATION .)
  if(APPLE)
    set(helper_destination VideoEditor.app/Contents/MacOS/tools)
  else()
    set(helper_destination ${CMAKE_INSTALL_BINDIR}/tools)
  endif()
  foreach(tool ffmpeg yt-dlp deno)
    set(helper "${VIDEO_EDITOR_TOOLS_DIR}/${tool}${CMAKE_EXECUTABLE_SUFFIX}")
    if(NOT EXISTS "${helper}")
      message(FATAL_ERROR "Missing ${helper}; run scripts/setup-tools.py first")
    endif()
    install(PROGRAMS "${helper}" DESTINATION ${helper_destination})
  endforeach()
  install(FILES "${VIDEO_EDITOR_TOOLS_DIR}/ffmpeg.LICENSE" "${VIDEO_EDITOR_TOOLS_DIR}/ffmpeg.README"
    DESTINATION ${helper_destination})
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_target_properties(VideoEditor PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    install(FILES packaging/video-editor.desktop DESTINATION share/applications)
    install(FILES packaging/video-editor.svg DESTINATION share/icons/hicolor/scalable/apps)
    set(deploy_options POST_INCLUDE_REGEXES "/lib.*[.]so([.]|$)"
      PRE_EXCLUDE_REGEXES "^ld-linux.*[.]so[.]")
    foreach(library c m mvec pthread dl rt resolv util anl BrokenLocale thread_db)
      list(APPEND deploy_options "^lib${library}[.]so[.]")
    endforeach()
    list(APPEND deploy_options "^libnss_.*[.]so[.]")
  endif()
  qt_generate_deploy_app_script(TARGET VideoEditor OUTPUT_SCRIPT deploy_script NO_TRANSLATIONS ${deploy_options})
  install(SCRIPT ${deploy_script})
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(editor_tests tests/editor_tests.cpp)
  set_target_properties(editor_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(editor_tests PRIVATE editor_core Qt6::Test)
  add_test(NAME editor_tests COMMAND ${CMAKE_COMMAND}
    "-DTEST_EXECUTABLE=$<TARGET_FILE:editor_tests>"
    "-DTEST_LOG=${CMAKE_CURRENT_BINARY_DIR}/test-results.txt"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/RunEditorTests.cmake")
  set_tests_properties(editor_tests PROPERTIES TIMEOUT 300
    ENVIRONMENT "QT_MEDIA_BACKEND=ffmpeg;QT_QPA_PLATFORM=offscreen")
endif()
